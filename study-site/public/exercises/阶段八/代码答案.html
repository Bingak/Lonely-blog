/* 答案_练习2_头文件守卫.c —— 阶段八练习2参考代码（单文件自包含版） * 编译：gcc -Wall -Wextra -std=c99 答案_练习2_头文件守卫.c -o a8_p2.exe * 运行：.\a8_p2.exe * * 单文件怎么演示「重复包含」？ * 把下面这一段当成 mylib.h 的内容，原样粘贴两遍：#include 的本质就是 * 预处理阶段的文本粘贴。第一遍让守卫宏生效，第二遍被 #ifndef 整段跳过。 * 真实工程的三文件写法见文件末尾注释。 */ #include <stdio.h> /* ══════════ 伪头文件 mylib.h：第 1 次粘贴 ══════════ */ #ifndef MYLIB_H #define MYLIB_H #define MYLIB_VERSION 1 #define MYLIB_MAX(a, b) (((a) > (b)) ? (a) : (b)) typedef struct { int x; int y; } MyPoint; extern int mylib_calls; /* 声明：定义在下面的「伪 .c」里 */ int mylib_manhattan(const MyPoint *p); void mylib_print_point(const MyPoint *p); #endif /* MYLIB_H */ /* ══════════ 伪头文件 mylib.h：第 2 次粘贴（模拟重复 include）══════════ */ #ifndef MYLIB_H #define MYLIB_H #define MYLIB_VERSION 1 #define MYLIB_MAX(a, b) (((a) > (b)) ? (a) : (b)) typedef struct { int x; int y; } MyPoint; extern int mylib_calls; int mylib_manhattan(const MyPoint *p); void mylib_print_point(const MyPoint *p); #endif /* MYLIB_H */ /* ══════════ 伪 mylib.c：真正的定义全程序只有一份 ══════════ */ int mylib_calls = 0; int mylib_manhattan(const MyPoint *p) { int ax = (p->x < 0) ? -p->x : p->x; /* |x| */ int ay = (p->y < 0) ? -p->y : p->y; /* |y| */ mylib_calls++; return ax + ay; /* 曼哈顿距离 = |x| + |y| */ } void mylib_print_point(const MyPoint *p) { printf("点 (%d, %d)\n", p->x, p->y); mylib_calls++; } int main(void) { MyPoint a = {-3, 4}; MyPoint b = {5, -2}; printf("MYLIB_VERSION = %d（两次包含只保留一份定义）\n", MYLIB_VERSION); printf("MYLIB_MAX(3, 9) = %d\n", MYLIB_MAX(3, 9)); mylib_print_point(&a); printf("到原点的曼哈顿距离 = %d\n", mylib_manhattan(&a)); printf("MYLIB_MAX(-1, -5) = %d（宏参数带括号，负数才不出错）\n", MYLIB_MAX(-1, -5)); mylib_print_point(&b); printf("到原点的曼哈顿距离 = %d\n", mylib_manhattan(&b)); printf("两个函数共被调用 %d 次（含打印）\n", mylib_calls); return 0; } /* 守卫为什么有效： * 第 1 次粘贴：MYLIB_H 未定义 → #ifndef 成立 → 定义 MYLIB_H，内容生效； * 第 2 次粘贴：MYLIB_H 已定义 → #ifndef 不成立 → 整段跳到 #endif。 * 把两处三行守卫删掉后，GCC 16 实测报： * error: conflicting types for 'MyPoint'; have 'struct <anonymous>' * note: previous declaration of 'MyPoint' with type 'MyPoint' * * 真实工程拆成三个文件： * mylib.h：守卫 + 宏 + 类型 + 函数原型 + extern 变量声明（只有声明） * mylib.c：#include "mylib.h" + 变量定义 + 函数定义 * main.c ：#include "mylib.h" + 调用（可以 include 两次，守卫会挡住） * 编译：gcc -Wall -Wextra -std=c99 main.c mylib.c -o app.exe * * #pragma once 与守卫的取舍： * 守卫：C 标准写法，任何编译器都认；缺点是三行样板，且宏名要全工程唯一 * （同名守卫会让后一个头文件被整段跳过——这个坑很隐蔽）。 * #pragma once：一行搞定，不会重名；但它不是 C 标准，靠「同一文件路径」 * 判断，遇到硬链接/复制出来的同名副本可能失效。 * 结论：作业、跨编译器、给别人看的代码用守卫；自己项目图省事可以用 * #pragma once，但要知道它依赖编译器扩展。 */ /* 预期输出（本机实测）： * MYLIB_VERSION = 1（两次包含只保留一份定义） * MYLIB_MAX(3, 9) = 9 * 点 (-3, 4) * 到原点的曼哈顿距离 = 7 * MYLIB_MAX(-1, -5) = -1（宏参数带括号，负数才不出错） * 点 (5, -2) * 到原点的曼哈顿距离 = 7 * 两个函数共被调用 4 次（含打印） */

/* 答案_练习4_无库字符串工具.c —— 阶段八练习4参考代码 * 编译：gcc -Wall -Wextra -std=c99 答案_练习4_无库字符串工具.c -o a8_p4.exe * 运行：.\a8_p4.exe * * 本文件教什么：不 include <string.h>，用指针手写 5 个字符串函数。 * 这 5 个函数是入队考核「禁用库函数」的直系积木：考核里处理输入、 * 比较命令、拼接提示，全靠它们。实现全部使用指针，函数体内不出现 [ ]。 */ #include <stdio.h> int my_strlen(const char *s); int my_strcmp(const char *a, const char *b); char *my_strcpy(char *dest, const char *src); char *my_strcat(char *dest, const char *src); const char *my_strchr(const char *s, char ch); int main(void) { char buf[64]; char cat[64] = "Robot "; const char *code = "UPRE2026"; const char *pos; printf("== my_strlen ==\n"); printf("空串长度 = %d（期望 0）\n", my_strlen("")); printf("UPRE2026 长度 = %d（期望 8）\n", my_strlen("UPRE2026")); printf("\n== my_strcmp ==\n"); printf("apple vs apple = %d（期望 0）\n", my_strcmp("apple", "apple")); printf("app vs apple = %d（期望负数）\n", my_strcmp("app", "apple")); printf("banana vs apple = %d（期望正数）\n", my_strcmp("banana", "apple")); printf("Apple vs apple = %d（大写字母编码更小）\n", my_strcmp("Apple", "apple")); printf("\n== my_strcpy ==\n"); my_strcpy(buf, "UPRE2026"); printf("复制后 buf = %s，长度 = %d\n", buf, my_strlen(buf)); printf("\n== my_strcat ==\n"); my_strcat(cat, "Lab"); printf("拼接后 cat = %s，长度 = %d\n", cat, my_strlen(cat)); printf("\n== my_strchr ==\n"); pos = my_strchr(code, '2'); if (pos != NULL) { printf("找 2：偏移 = %d，从它开始的尾部 = %s\n", (int)(pos - code), pos); } pos = my_strchr(code, 'Z'); printf("找 Z：%s（期望 NULL）\n", (pos == NULL) ? "没找到" : "找到了"); pos = my_strchr(code, '\0'); printf("找结尾 0：偏移 = %d（期望 8，指向 '\\0'）\n", (int)(pos - code)); return 0; } int my_strlen(const char *s) { const char *p = s; while (*p != '\0') { p++; } return (int)(p - s); /* 指针相减 = 中间隔了几个字符 */ } int my_strcmp(const char *a, const char *b) { while (*a != '\0' && *a == *b) { a++; b++; } return (int)(unsigned char)*a - (int)(unsigned char)*b; } char *my_strcpy(char *dest, const char *src) { char *d = dest; while (*src != '\0') { *d = *src; /* 逐字符搬 */ d++; src++; } *d = '\0'; /* 手动补结尾，最容易漏的一步 */ return dest; } char *my_strcat(char *dest, const char *src) { char *d = dest; while (*d != '\0') { /* 先走到 dest 的末尾 */ d++; } while (*src != '\0') { /* 再从末尾往后搬 src */ *d = *src; d++; src++; } *d = '\0'; return dest; } const char *my_strchr(const char *s, char ch) { while (*s != '\0') { if (*s == ch) { return s; } s++; } if (ch == '\0') { return s; /* 找结尾符：返回指向 '\0' 的指针 */ } return NULL; } /* 容量责任（危险写法只放在注释里，别真跑）： * char small[8]; * my_strcpy(small, "0123456789"); ← 越界写！ * 标准库的 strcpy 也不检查目标容量，谁分配谁负责： * 复制要 strlen(src) + 1 字节；拼接要 strlen(dest) + strlen(src) + 1 字节。 */ /* 预期输出（本机实测）： * == my_strlen == * 空串长度 = 0（期望 0） * UPRE2026 长度 = 8（期望 8） * * == my_strcmp == * apple vs apple = 0（期望 0） * app vs apple = -108（期望负数） * banana vs apple = 1（期望正数） * Apple vs apple = -32（大写字母编码更小） * * == my_strcpy == * 复制后 buf = UPRE2026，长度 = 8 * * == my_strcat == * 拼接后 cat = Robot Lab，长度 = 9 * * == my_strchr == * 找 2：偏移 = 4，从它开始的尾部 = 2026 * 找 Z：没找到（期望 NULL） * 找结尾 0：偏移 = 8（期望 8，指向 '\0'） */

/* student.c —— 阶段八练习5答案：学生成绩管理的实现文件 * 编译：gcc -Wall -Wextra -std=c99 -I. -c student.c -o student.o * 链接：gcc -Wall -Wextra -std=c99 main.c student.c -o grade.exe * 运行：.\grade.exe * * 要点：第一行 #include "student.h" 让编译器拿原型核对定义； * 所有函数只通过参数拿数据，不用全局数组（模块化的基本要求）。 */ #include <stdio.h> #include "student.h" /* 录入：返回实际录入人数；人数非法或输入格式错时提前返回 */ int input_students(Student *list, int max) { int n, i; printf("要录入几个学生（1~%d）：", max); if (scanf("%d", &n) != 1 || n < 1 || n > max) { printf("人数无效。\n"); return 0; } for (i = 0; i < n; i++) { printf("第 %d 个：姓名 学号 三门成绩（空格分隔）：", i + 1); /* %31s 限制宽度，最多写 31 个字符 + 结尾 0，正好放进 name[32] */ if (scanf("%31s %d %d %d %d", list[i].name, &list[i].id, &list[i].scores[0], &list[i].scores[1], &list[i].scores[2]) != 5) { printf("输入格式错误，停止录入。\n"); return i; } calc_average(&list[i]); /* 录完立刻算平均分 */ } return n; } /* 算单个学生的平均分：直接改结构体里的 average（传指针才改得到原件） */ void calc_average(Student *s) { int i, sum = 0; for (i = 0; i < COURSE_NUM; i++) { sum += s->scores[i]; } s->average = (double)sum / COURSE_NUM; } /* 打印一个学生：姓名左对齐，平均分保留两位小数 */ void print_student(const Student *s) { printf("%-10s %8d 平均 %6.2f 成绩（%d, %d, %d）\n", s->name, s->id, s->average, s->scores[0], s->scores[1], s->scores[2]); } /* 打印全部：只读，所以参数加 const */ void print_all(const Student *list, int n) { int i; for (i = 0; i < n; i++) { print_student(&list[i]); } } /* 按平均分从高到低排序：选择排序，交换整个结构体 */ void sort_by_average(Student *list, int n) { int i, j, best; Student tmp; for (i = 0; i < n - 1; i++) { best = i; for (j = i + 1; j < n; j++) { if (list[j].average > list[best].average) { best = j; } } if (best != i) { tmp = list[i]; /* 结构体可以整体赋值（阶段六） */ list[i] = list[best]; list[best] = tmp; } } }

/* 答案_练习7_函数指针菜单.c —— 阶段八练习7参考代码 * 编译：gcc -Wall -Wextra -std=c99 答案_练习7_函数指针菜单.c -o a8_p7.exe * 运行：.\a8_p7.exe * 输入示例：help / hello / sum / 100 / upper / upre robot / pi / greet / quit * * 要点：命令表把「命令名 → 函数」绑在一起，主循环只做「查表 + 调用」。 * 加命令 = 表里加一行 + 写一个函数，主循环一行不动。 * 比 switch 好：① 加命令不改主循环；② 名字/函数/说明集中一张表， * help 自动生成；③ 表可被数据驱动（读配置、批量执行），switch 做不到。 */ #include <stdio.h> #include <string.h> #define LINE_MAX 64 #define COMMAND_COUNT ((int)(sizeof g_commands / sizeof g_commands[0])) typedef void (*CommandFn)(void); typedef struct { const char *name; CommandFn run; const char *help; } Command; static void cmd_hello(void); static void cmd_sum(void); static void cmd_upper(void); static void cmd_pi(void); static void cmd_help(void); if (!read_line(line, sizeof line)) { printf("\n输入结束，退出。\n"); break; } if (line[0] == '\0') continue; if (strcmp(line, "quit") == 0) { printf("再见。\n"); break; } found = 0; for (i = 0; i < COMMAND_COUNT; i++) { if (strcmp(line, g_commands[i].name) == 0) { g_commands[i].run(); /* 查表 + 调用，没有 switch */ found = 1; break; } } if (!found) printf("未知命令：%s（输入 help 查看）\n", line); } return 0; } static void cmd_hello(void) { printf("你好，这里是函数指针菜单。\n"); } static void cmd_sum(void) { int n, i, sum = 0; if (!read_int("输入 n：", &n) || n < 1) { printf("输入无效。\n"); return; } for (i = 1; i <= n; i++) sum += i; printf("1 到 %d 的和 = %d\n", n, sum); } static void cmd_upper(void) { char line[LINE_MAX]; char *p; printf("输入一行小写文字："); if (!read_line(line, sizeof line)) return; for (p = line; *p != '\0'; p++) if (*p >= 'a' && *p <= 'z') *p = (char)(*p - 'a' + 'A'); printf("转成大写：%s\n", line); } static void cmd_pi(void) { printf("pi 约等于 3.14159\n");
