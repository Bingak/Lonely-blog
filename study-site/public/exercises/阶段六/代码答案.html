/* 答案_练习2_矩形面积与包含.c —— 阶段六练习2参考代码 * 编译：gcc -Wall -Wextra 答案_练习2_矩形面积与包含.c -o a6_p2.exe * 要点：矩形的左下角 + 宽高表示法；contains 用 1/0 作真假标志（int）。 * 含边界：用 <= 与 >=，边界点算在内。 */ #include <stdio.h> struct Point { double x; double y; }; struct Rect { double x; /* 左下角横坐标 */ double y; /* 左下角纵坐标 */ double w; /* 宽 */ double h; /* 高 */ }; double area(struct Rect r); int contains(struct Rect r, struct Point p); int main(void) { struct Rect r = {0.0, 0.0, 10.0, 5.0}; printf("矩形面积 = %.1f（应得 50.0）\n", area(r)); struct Point a = {5.0, 2.0}; /* 内部 */ struct Point b = {10.0, 5.0}; /* 右上角：边界，算在内 */ struct Point c = {11.0, 2.0}; /* 外部 */ printf("(5,2) 在矩形内？%d（应得 1）\n", contains(r, a)); printf("(10,5) 在矩形内？%d（应得 1，含边界）\n", contains(r, b)); printf("(11,2) 在矩形内？%d（应得 0）\n", contains(r, c)); return 0; } double area(struct Rect r) { return r.w * r.h; } int contains(struct Rect r, struct Point p) { return p.x >= r.x && p.x <= r.x + r.w && p.y >= r.y && p.y <= r.y + r.h; } /* 预期输出： * 矩形面积 = 50.0（应得 50.0） * (5,2) 在矩形内？1（应得 1） * (10,5) 在矩形内？1（应得 1，含边界） * (11,2) 在矩形内？0（应得 0） */
CODE答案_练习4_日期比较.c
/* 答案_练习4_日期比较.c —— 阶段六练习4参考代码 * 编译：gcc -Wall -Wextra 答案_练习4_日期比较.c -o a6_p4.exe * 要点：年→月→日字典序逐成员比较，不许换算总天数； * earliest 复用擂台法（阶段四），比较换成 date_cmp。 */ #include <stdio.h> struct Date { int year; int month; int day; }; int date_cmp(struct Date a, struct Date b); struct Date earliest(const struct Date arr[], int n); int main(void) { struct Date d1 = {2026, 10, 25}; struct Date d2 = {2026, 10, 26}; /* 同年同月不同日 */ struct Date d3 = {2026, 9, 25}; /* 同年不同月 */ struct Date d4 = {2027, 1, 1}; /* 不同年 */ struct Date d5 = {2026, 10, 25}; /* 完全相同 */ printf("cmp(d1,d2) = %2d（应得 -1）\n", date_cmp(d1, d2)); printf("cmp(d1,d3) = %2d（应得 1）\n", date_cmp(d1, d3)); printf("cmp(d1,d4) = %2d（应得 -1）\n", date_cmp(d1, d4)); printf("cmp(d1,d5) = %2d（应得 0）\n", date_cmp(d1, d5)); struct Date list[4] = {{2026, 10, 25}, {2025, 12, 31}, {2026, 1, 1}, {2026, 10, 24}}; struct Date min = earliest(list, 4); printf("最早的一天：%04d-%02d-%02d（应得 2025-12-31）\n", min.year, min.month, min.day); return 0; } /* 逐成员比较：先分胜负的直接返回，平局才往下看 */ int date_cmp(struct Date a, struct Date b) { if (a.year != b.year) return a.year < b.year ? -1 : 1; if (a.month != b.month) return a.month < b.month ? -1 : 1; if (a.day != b.day) return a.day < b.day ? -1 : 1; return 0; } /* 擂台法：结构体整体赋值当「擂主交接」（值语义，§4） */ struct Date earliest(const struct Date arr[], int n) { struct Date best = arr[0]; int i; for (i = 1; i < n; i++) { if (date_cmp(arr[i], best) < 0) { best = arr[i]; /* 整个盒子一起换手 */ } } return best; /* 返回也是整个复印回去 */ } /* 预期输出： * cmp(d1,d2) = -1（应得 -1） * cmp(d1,d3) = 1（应得 1） * cmp(d1,d4) = -1（应得 -1） * cmp(d1,d5) = 0（应得 0） * 最早的一天：2025-12-31（应得 2025-12-31） */
CODE答案_练习5_按年份排序.c
/* 答案_练习5_按年份排序.c —— 阶段六练习5参考代码 * 编译：gcc -Wall -Wextra 答案_练习5_按年份排序.c -o a6_p5.exe * 要点：冒泡排序；相邻交换用结构体整体赋值（临时变量倒手， * 与 int 交换一模一样——这就是值语义的威力）。 */ #include <stdio.h> #define TITLE_LEN 40 struct Book { char title[TITLE_LEN]; int year; double price; }; void print_all(const char *label, const struct Book shelf[], int n); void sort_by_year(struct Book shelf[], int n); int main(void) { struct Book shelf[5] = { {"C Primer Plus", 2015, 128.00}, {"The C Programming Language", 1988, 45.00}, {"C Traps and Pitfalls", 1989, 39.00}, {"Pointers on C", 1997, 55.00}, {"Expert C Programming", 1994, 65.00} }; int n = 5; print_all("排序前", shelf, n); sort_by_year(shelf, n); print_all("排序后", shelf, n); return 0; } void sort_by_year(struct Book shelf[], int n) { int i, j; for (i = 0; i < n - 1; i++) { /* 冒泡：n-1 趟 */ for (j = 0; j < n - 1 - i; j++) { /* 每趟把最大的推到末尾 */ if (shelf[j].year > shelf[j + 1].year) { struct Book t = shelf[j]; /* 整体赋值倒手 */ shelf[j] = shelf[j + 1]; shelf[j + 1] = t; } } } } void print_all(const char *label, const struct Book shelf[], int n) { int i; printf("%s：\n", label); for (i = 0; i < n; i++) { printf(" %-28s %d 年 %.2f 元\n", shelf[i].title, shelf[i].year, shelf[i].price); } } /* 预期输出： * 排序前： * C Primer Plus 2015 年 128.00 元 * The C Programming Language 1988 年 45.00 元 * C Traps and Pitfalls 1989 年 39.00 元 * Pointers on C 1997 年 55.00 元 * Expert C Programming 1994 年 65.00 元 * 排序后： * The C Programming Language 1988 年 45.00 元 * C Traps and Pitfalls 1989 年 39.00 元 * Expert C Programming 1994 年 65.00 元 * Pointers on C 1997 年 55.00 元 * C Primer Plus 2015 年 128.00 元 */
CODE答案_练习6_union字节观察.c
/* 答案_练习6_union字节观察.c —— 阶段六练习6参考代码 * 编译：gcc -Wall -Wextra 答案_练习6_union字节观察.c -o a6_p6.exe * 要点：先纸上预测再运行验证；小端 = 最低字节存在最前面。 * -1 的四个字节全是 FF（补码，阶段七回收这个伏笔）。 */ #include <stdio.h> union IntViewer { int whole; unsigned char bytes[4]; }; union ShortViewer { short s; unsigned char b[2]; }; static void show_int(union IntViewer v, const char *note) { printf("%-22s bytes = %02X %02X %02X %02X\n", note, v.bytes[0], v.bytes[1], v.bytes[2], v.bytes[3]); } int main(void) { union IntViewer v; /* ① 预测：小端机器，0x00FF00FF 最低字节 FF 在最前面， 即 bytes = FF 00 FF 00 */ v.whole = 0x00FF00FF; show_int(v, "0x00FF00FF"); /* ② 256 = 0x00000100，次低字节是 01： 预测 bytes = 00 01 00 00 */ v.whole = 256; show_int(v, "256"); /* ③ -1：先运行再看——四个字节全是 FF，为什么留给阶段七 */ v.whole = -1; show_int(v, "-1"); /* ④ short 版：0x0102 → 预测 b = 02 01 */ union ShortViewer w; w.s = 0x0102; printf("%-22s b = %02X %02X\n", "short 0x0102", w.b[0], w.b[1]); return 0; } /* 预期输出： * 0x00FF00FF bytes = FF 00 FF 00 * 256 bytes = 00 01 00 00 * -1 bytes = FF FF FF FF * short 0x0102 b = 02 01 */
CODE答案_练习7_枚举菜单重构.c
/* 答案_练习7_枚举菜单重构.c —— 阶段六练习7参考代码 * 编译：gcc -Wall -Wextra 答案_练习7_枚举菜单重构.c -o a6_p7.exe * 要点：魔法数字 0/1/2/3 → enum 菜单名；if-else 链 → switch； * 换算封装成函数（阶段三）；scanf 检查返回值（阶段二）。 */ #include <stdio.h> enum Menu { MENU_EXIT = 0, MENU_KM2MI, MENU_C2F, MENU_KG2LB }; double km_to_mi(double km); double c_to_f(double c); double kg_to_lb(double kg); void print_menu(void); int main(void) { int choice = -1; double value; while (choice != MENU_EXIT) { print_menu(); if (scanf("%d", &choice) != 1) { /* 阶段二：检查返回值 */ while (getchar() != '\n') { } /* 清空缓冲区里的垃圾 */ printf("输入无效：请输入数字\n"); continue; } switch (choice) { /* case 全部用枚举名 */ case MENU_EXIT: printf("再见\n"); break; case MENU_KM2MI: printf("输入千米："); if (scanf("%lf", &value) != 1) return 1; printf("%.2f km = %.2f mi\n", value, km_to_mi(value)); break; case MENU_C2F: printf("输入摄氏温度："); if (scanf("%lf", &value) != 1) return 1; printf("%.2f C = %.2f F\n", value, c_to_f(value)); break; case MENU_KG2LB: printf("输入千克："); if (scanf("%lf", &value) != 1) return 1; printf("%.2f kg = %.2f lb\n", value, kg_to_lb(value)); break; default: printf("无效选项\n"); break; } } return 0; } void print_menu(void) { printf("%d 退出 %d 千米→英里 %d 摄氏→华氏 %d 千克→磅\n", MENU_EXIT, MENU_KM2MI, MENU_C2F, MENU_KG2LB); } double km_to_mi(double km) { return km * 0.6214; } double c_to_f(double c) { return c * 1.8 + 32.0; } double kg_to_lb(double kg) { return kg * 2.2046; } /* 预期输出（输入序列：1 → 100 → 2 → 25 → 3 → 60 → 9 → 0）： * 0 退出 1 千米→英里 2 摄氏→华氏 3 千克→磅 * 输入千米：100.00 km = 62.14 mi * 0 退出 1 千米→英里 2 摄氏→华氏 3 千克→磅 * 输入摄氏温度：25.00 C = 77.00 F * 0 退出 1 千米→英里 2 摄氏→华氏 3 千克→磅 * 输入千克：60.00 kg = 132.28 lb * 0 退出 1 千米→英里 2 摄氏→华氏 3 千克→磅 * 无效选项 * 0 退出 1 千米→英里 2 摄氏→华氏 3 千克→磅 * 再见 */
