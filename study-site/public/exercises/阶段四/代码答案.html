/* 答案_练习1_数组逆置.c —— 阶段四练习1参考代码 * 编译：gcc -Wall -Wextra 答案_练习1_数组逆置.c -o a4_p1.exe * 要点：两头互换向中间夹，i 只走到 n/2；边界 n=0、n=1 天然安全。 */ #include <stdio.h> void reverse(int arr[], int n); void print_array(const char *label, const int arr[], int n); int main(void) { int a[7] = {1, 2, 3, 4, 5, 6, 7}; int n = (int)(sizeof(a) / sizeof(a[0])); print_array("原数组", a, n); reverse(a, n); print_array("逆置后", a, n); reverse(a, n); /* 再逆置 = 回到原样（自检） */ print_array("再逆置", a, n); /* 边界自测 */ int single[1] = {42}; reverse(single, 1); printf("单元素逆置后：%d\n", single[0]); reverse(single, 0); /* n=0：循环一次都不进，安全 */ printf("n=0 逆置后：%d\n", single[0]); return 0; } void reverse(int arr[], int n) { int i; for (i = 0; i < n / 2; i++) { /* 换一半就够；n=0 或 1 时循环不执行 */ int t = arr[i]; arr[i] = arr[n - 1 - i]; arr[n - 1 - i] = t; } } void print_array(const char *label, const int arr[], int n) { int i; printf("%-6s: [", label); for (i = 0; i < n; i++) { printf("%d", arr[i]); if (i < n - 1) printf(" "); } printf("]\n"); } /* 预期输出： * 原数组 : [1 2 3 4 5 6 7] * 逆置后 : [7 6 5 4 3 2 1] * 再逆置 : [1 2 3 4 5 6 7] * 单元素逆置后：42 * n=0 逆置后：42 */

/* 答案_练习2_成绩统计.c —— 阶段四练习2参考代码 * 编译：gcc -Wall -Wextra 答案_练习2_成绩统计.c -o a4_p2.exe * 要点：擂台法返回"下标"而非"值"（信息更全）；double 数组的函数参数。 */ #include <stdio.h> double average(const double arr[], int n); int max_index(const double arr[], int n); int main(void) { double scores[6] = {78.5, 92.0, 66.5, 88.0, 92.0, 59.5}; int n = (int)(sizeof(scores) / sizeof(scores[0])); int best = max_index(scores, n); printf("平均分 = %.2f\n", average(scores, n)); printf("最高分 = %.1f，在下标 %d（第 %d 个）\n", scores[best], best, best + 1); printf("注：92.0 出现两次，保留先出现的（下标 1）\n"); return 0; } double average(const double arr[], int n) { if (n <= 0) { return 0.0; /* 防御：空数组约定返回 0 */ } double total = 0.0; int i; for (i = 0; i < n; i++) { total += arr[i]; } return total / n; } int max_index(const double arr[], int n) { int best = 0; int i; for (i = 1; i < n; i++) { if (arr[i] > arr[best]) { /* 严格大于：并列保留先来的 */ best = i; } } return best; } /* 预期输出： * 平均分 = 79.42 * 最高分 = 92.0，在下标 1（第 2 个） * 注：92.0 出现两次，保留先出现的（下标 1） */

/* 答案_练习4_二进制转换.c —— 阶段四练习4参考代码 * 编译：gcc -Wall -Wextra 答案_练习4_二进制转换.c -o a4_p4.exe * 要点：除2取余存数组 + 返回位数 + 倒序输出（数组带出数据的又一实战） * 模式与阶段三 num_to_digits 同构——能力是可迁移的。 */ #include <stdio.h> int to_binary(int n, int bits[]); /* 返回位数，bits 从低位往高位存 */ int main(void) { int bits[32]; /* int 最多 32 位，开 32 格永不越界 */ int tests[] = {0, 1, 5, 13, 255, 1000000}; int i; for (i = 0; i < 6; i++) { int count = to_binary(tests[i], bits); printf("%8d 的二进制 = ", tests[i]); int j; for (j = count - 1; j >= 0; j--) { /* 倒序输出：高位在前 */ printf("%d", bits[j]); } printf("\n"); } return 0; } int to_binary(int n, int bits[]) { if (n == 0) { bits[0] = 0; /* 边界：0 单独处理，否则一次都不进循环 */ return 1; } int count = 0; while (n > 0) { bits[count] = n % 2; /* 余数从低位往高位逐个入数组 */ count++; n /= 2; } return count; /* 返回位数，数组内容由调用方倒序读 */ } /* 预期输出： * 0 的二进制 = 0 * 1 的二进制 = 1 * 5 的二进制 = 101 * 13 的二进制 = 1101 * 255 的二进制 = 11111111 * 1000000 的二进制 = 11110100001001000000 */
CODE答案_练习5_凯撒密码.c
/* 答案_练习5_凯撒密码.c —— 阶段四练习5参考代码 * 编译：gcc -Wall -Wextra 答案_练习5_凯撒密码.c -o a4_p5.exe * 运行后输入：Attack at dawn! 预期输出：Dwwdfn dw gdzq! * 要点：字母内循环 +3（x→a）；大小写分别处理；非字母原样保留。 * 公式：c-'a' 变成 0~25，加 3 后对 26 取模，再加回 'a' —— "绕圈"的标准写法。 */ #include <stdio.h> int my_strlen(const char s[]); int main(void) { char text[101]; char encrypted[101]; printf("请输入要加密的明文："); if (fgets(text, sizeof(text), stdin) == NULL) { return 1; } /* 清理 fgets 带进来的换行 */ int len = my_strlen(text); if (len > 0 && text[len - 1] == '\n') { text[len - 1] = '\0'; len--; } int i; for (i = 0; i < len; i++) { char c = text[i]; if (c >= 'a' && c <= 'z') { encrypted[i] = (char)((c - 'a' + 3) % 26 + 'a'); /* 小写绕圈 */ } else if (c >= 'A' && c <= 'Z') { encrypted[i] = (char)((c - 'A' + 3) % 26 + 'A'); /* 大写绕圈 */ } else { encrypted[i] = c; /* 非字母原样保留 */ } } encrypted[len] = '\0'; printf("密文：%s\n", encrypted); return 0; } int my_strlen(const char s[]) { int i = 0; while (s[i] != '\0') i++; return i; } /* 预期输出： * 请输入要加密的明文：Attack at dawn! * 密文：Dwwdfn dw gdzq! * 手工验证 'x'：( 'x'-'a' + 3 ) % 26 = (23+3)%26 = 0 → 'a' ✔ 绕圈成功 */

/* 答案_练习6_字符分类器.c —— 阶段四练习6参考代码 * 编译：gcc -Wall -Wextra 答案_练习6_字符分类器.c -o a4_p6.exe * 运行后输入任意一行观察统计。 * 本题约定：先把 fgets 的 '\n' 清理掉，再统计——所以 '\n' 不算「其他」。 */ #include <stdio.h> int my_strlen(const char s[]); int main(void) { char line[201]; printf("请输入一行文字："); if (fgets(line, sizeof(line), stdin) == NULL) { return 1; } /* 清理换行（决定它不参与分类） */ int len = my_strlen(line); if (len > 0 && line[len - 1] == '\n') { line[len - 1] = '\0'; len--; } int upper = 0, lower = 0, digits = 0, spaces = 0, others = 0; /* 计数器组归零 */ int i; for (i = 0; i < len; i++) { char c = line[i]; if (c >= 'A' && c <= 'Z') { upper++; } else if (c >= 'a' && c <= 'z') { lower++; } else if (c >= '0' && c <= '9') { digits++; } else if (c == ' ') { spaces++; } else { others++; /* 标点符号、逗号等 */ } } printf("大写 %d、小写 %d、数字 %d、空格 %d、其他 %d\n", upper, lower, digits, spaces, others); return 0; } int my_strlen(const char s[]) { int i = 0; while (s[i] != '\0') i++; return i; } /* 预期输出（输入 Hello, C Language 2026!）： * 请输入一行文字：Hello, C Language 2026! * 大写 3、小写 12、数字 4、空格 3、其他 2 * 验证：H,C,L 三个大写；空格3个：Hello后/C后/2026前；逗号感叹号=其他2 */
CODE答案_练习7_单词计数器.c
/* 答案_练习7_单词计数器.c —— 阶段四练习7参考代码 * 编译：gcc -Wall -Wextra 答案_练习7_单词计数器.c -o a4_p7.exe * 核心：状态切换思想 —— "当前是字母且上一格不是字母" = 新单词开始。 * 与连续模式计数是孪生兄弟：一个数「模式」，一个数「模式的开头」。 * 单词定义：连续的非空格字符（don't 算 1 个；Hello, 算 1 个——长度含标点）。 */ #include <stdio.h> int my_strlen(const char s[]); int main(void) { char line[201]; printf("请输入一行英文："); if (fgets(line, sizeof(line), stdin) == NULL) { return 1; } int len = my_strlen(line); if (len > 0 && line[len - 1] == '\n') { line[len - 1] = '\0'; len--; } int word_count = 0; int best_len = 0; int best_start = 0; /* 最长单词的起点 */ int i; for (i = 0; i < len; i++) { char c = line[i]; char prev = (i == 0) ? ' ' : line[i - 1]; if (c != ' ') { if (prev == ' ') { /* 状态切换：上一格是空格 → 新单词诞生 */ word_count++; } /* 统计当前单词长度：从当前起点往后数（用 j 扫描） */ int j = i; while (j < len && line[j] != ' ') { j++; } if (j - i > best_len) { /* j-i = 这个单词的长度 */ best_len = j - i; best_start = i; } } } printf("单词个数 = %d，最长单词长度 = %d，它是：", word_count, best_len); int k; for (k = best_start; k < best_start + best_len; k++) { putchar(line[k]); } printf("\n"); return 0; } int my_strlen(const char s[]) { int i = 0; while (s[i] != '\0') i++; return i; } /* 预期输出（输入 To be or not to be that is the question）： * 请输入一行英文：To be or not to be that is the question * 单词个数 = 10，最长单词长度 = 8，它是：question *（自测边界：行首就是单词 / 连续多个空格 / 全空格行 / 空行—— * 后两种应该输出 0 个，试一试你的版本！） */
