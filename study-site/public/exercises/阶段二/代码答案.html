/* 答案_练习1_正序分解修正版.c —— 阶段二练习1参考代码 * 编译：gcc -Wall -Wextra 答案_练习1_正序分解修正版.c -o a2_p1.exe * 修正两处： * P1: ret 未初始化 → 定义即初始化 int ret = 0; * P2: while (num > 10) → while (num >= 10)（两位数的十位不再漏掉） * 自测：5 → "5"；23 → "2 3"；100 → "1 0 0"；12345 → "1 2 3 4 5" */ #include <stdio.h> int main(void) { int num; int ret = 0; /* ★ 修正点1：出生即初始化 */ printf("请输入数字："); if (scanf("%d", &num) != 1) { printf("输入不合法\n"); return 1; } while (num >= 10) { /* ★ 修正点2：>= 让两位数十位也参与 */ int digit = num % 10; ret = ret * 10 + digit; num /= 10; } ret = ret * 10 + num; /* 最高位也要并入反转数 */ /* ret 现在是"各位倒序"后的整数，逐位打印即得到正序输出 */ while (ret > 0) { printf("%d ", ret % 10); ret /= 10; } printf("\n"); return 0; } /* 推演示范（输入 23）： * 原版：23 > 10 成立 → ret=3, num=2；2 > 10 不成立退出 → 十位丢失 * 修正：23 >= 10 → ret=3, num=2；2 >= 10 不成立 → 退出循环后补并最高位 ret=32 * 打印 2 3 ✔ */

CODE答案_练习4_个税计算器.c
/* 答案_练习4_个税计算器.c —— 阶段二练习4参考代码 * 编译：gcc -Wall -Wextra 答案_练习4_个税计算器.c -o a2_p4.exe * 要点：边界收入税额必须连续。 * 收入 3000 → 税 0（两档在此相接） * 收入 12000 → 税 270（第二档满额：9000*3%=270；第三档在此相接） * 自测点：2999.99→0.00；3000→0.00；3000.01→0.00（差0.0003，四舍五入后为0） * 12000→270.00；12000.01→270.00+0.001≈270.00（连续！） */ #include <stdio.h> int main(void) { double income; printf("请输入月收入："); if (scanf("%lf", &income) != 1 || income < 0) { printf("输入不合法\n"); return 1; } double tax = 0.0; if (income <= 3000) { tax = 0.0; /* 第一档：免税 */ } else if (income <= 12000) { tax = (income - 3000) * 0.03; /* 第二档：超出3000部分*3% */ } else { tax = 270.0 + (income - 12000) * 0.10; /* 第三档：270 + 超出12000部分*10% */ } printf("应纳税：%.2f 元\n", tax); return 0; } /* 预期输出： * 输入 3000 → 应纳税：0.00 元 * 输入 12000 → 应纳税：270.00 元 * 输入 20000 → 应纳税：270 + 800 = 1070.00 元 */
CODE答案_练习5_水仙花无pow.c
/* 答案_练习5_水仙花无pow.c —— 阶段二练习5参考代码 * 编译：gcc -Wall -Wextra 答案_练习5_水仙花无pow.c -o a2_p5.exe * 要点：整数世界用整数工具。循环累乘替代 pow（浮点函数有截断风险）。 * 验证：n=3 时必须找出 153、370、371、407 全部四个。 */ #include <stdio.h> /* 整数幂：base 的 exp 次方（exp >= 0），返回 int */ int int_pow(int base, int exp) { int result = 1; /* 累乘器：出生即初始化为 1（乘法单位元） */ int i; for (i = 0; i < exp; i++) { result *= base; } return result; } int main(void) { int n; printf("请输入位数 n："); if (scanf("%d", &n) != 1 || n < 1) { printf("输入不合法\n"); return 1; } int start = 1; /* n 位数的下界：10^(n-1) */ int end; /* n 位数的上界（不含）：10^n */ int i; for (i = 1; i < n; i++) { start *= 10; } end = start * 10; for (i = start; i < end; i++) { int sum = 0; int tmp = i; /* 临时副本：分解时不动原数 i */ while (tmp > 0) { int d = tmp % 10; /* 取出当前最低位 */ sum += int_pow(d, n); tmp /= 10; } if (sum == i) { printf("%d\n", i); } } return 0; } /* 预期输出（n=3）： * 153 * 370 * 371 * 407 * 注意：这个文件出现了你第一个"自定义函数 int_pow"！ * 先跑起来感受一下，阶段三会讲清楚它的每一行——那正是任务三需要的知识。 */

/* 答案_练习6_单位换算菜单.c —— 阶段二练习6参考代码 * 编译：gcc -Wall -Wextra 答案_练习6_单位换算菜单.c -o a2_p6.exe * 骨架：do-while 主菜单 + switch 分支 + 输入合法性检查 * 这是"菜单循环"类程序的标准形态（考核任务骨架的训练版，具体任务请独立完成） */ #include <stdio.h> int main() { int choice; double value; do { printf("==== 单位换算 ====\n"); printf("1. 厘米 -> 英寸\n"); printf("2. 公斤 -> 磅\n"); printf("3. 摄氏 -> 华氏\n"); printf("0. 退出\n"); printf("请选择："); if (scanf("%d", &choice) != 1) { /* 输入非整数：清空缓冲区重来 */ while (getchar() != '\n') { } /* 吃掉本行剩余字符（阶段四详解） */ printf("请输入菜单里的数字！\n\n"); continue; /* 直接进入下一轮菜单显示 */ } switch (choice) { case 1: printf("输入厘米数："); if (scanf("%lf", &value) == 1) { printf("= %.4f 英寸\n\n", value / 2.54); } break; case 2: printf("输入公斤数："); if (scanf("%lf", &value) == 1) { printf("= %.4f 磅\n\n", value * 2.2046226); } break; case 3: printf("输入摄氏温度："); if (scanf("%lf", &value) == 1) { printf("= %.4f 华氏度\n\n", value * 9.0 / 5.0 + 32.0); } break; case 0: printf("已退出\n"); break; default: printf("没有这个选项！\n\n"); } } while (choice != 0); return 0; } /* 学法提示： * 1. 先跑通感受"菜单循环"的结构：显示→读入→分发→回到显示； * 2. 遮住代码，尝试凭记忆重写这个骨架（写不出就再看，直到3分钟能默写）； * 3. 思考：为什么 while 条件放在 do-while 的 while 里而不是中间？ */
