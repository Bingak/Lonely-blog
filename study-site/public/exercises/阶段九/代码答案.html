/* 答案_练习1_动态数组入门.c —— 阶段九练习1参考代码 * 编译（PowerShell）： * & "C:\Program Files (x86)\mingw64\bin\gcc.exe" -Wall -Wextra -std=c99 "答案_练习1_动态数组入门.c" -o a9_p1.exe * 运行（交互）：.\a9_p1.exe * 运行（管道自测）："5`n3 9 1 7 5" | .\a9_p1.exe * * 要点： * ① n 到运行期才读入，数组用 malloc 现开——栈上定长数组做不到 * ② malloc 后立刻判 NULL；free 后立刻把指针置 NULL * ③ 一个 malloc 对应一个 free，中途每个失败出口也要 free */ #include <stdio.h> #include <stdlib.h> int main(void) { int n; printf("请输入整数个数 n（1~1000）：\n"); if (scanf("%d", &n) != 1 || n < 1 || n > 1000) { printf("n 不合法，程序退出\n"); return 1; } /* 长度运行期才知道：用 n 算字节数，绝不写死 4 */ int *arr = malloc((size_t)n * sizeof(int)); if (arr == NULL) { printf("malloc 失败：内存不足\n"); return 1; } printf("请依次输入 %d 个整数：\n", n); int i; for (i = 0; i < n; i++) { if (scanf("%d", &arr[i]) != 1) { printf("输入不足，释放内存后退出\n"); free(arr); /* 失败出口也不能漏 free */ arr = NULL; return 1; } } printf("数组内容："); for (i = 0; i < n; i++) { printf("%d ", arr[i]); } printf("\n"); long long sum = 0; /* 防溢出：1000 个大 int 相加可能超 32 位 */ int max = arr[0]; int max_index = 0; for (i = 0; i < n; i++) { sum += arr[i]; if (arr[i] > max) { /* 擂台法：严格大于才换擂主 */ max = arr[i]; max_index = i; } } printf("总和 = %lld\n", sum); printf("最大值 = %d（第 %d 个元素，下标 %d）\n", max, max_index + 1, max_index); printf("逆序打印："); for (i = n - 1; i >= 0; i--) { /* 倒着遍历，不用另开数组 */ printf("%d ", arr[i]); } printf("\n"); free(arr); /* 唯一的 free，对应唯一的 malloc */ arr = NULL; printf("free 后 arr = %p，一个 malloc 对应一个 free。\n", (void *)arr); return 0; } /* 预期输出（管道实测：先输入 5，再输入 3 9 1 7 5）： * 请输入整数个数 n（1~1000）： * 请依次输入 5 个整数： * 数组内容：3 9 1 7 5 * 总和 = 25 * 最大值 = 9（第 2 个元素，下标 1） * 逆序打印：5 7 1 9 3 * free 后 arr = 0x0，一个 malloc 对应一个 free。 */

/* 答案_练习2_calloc计数.c —— 阶段九练习2参考代码 * 编译（PowerShell）： * & "C:\Program Files (x86)\mingw64\bin\gcc.exe" -Wall -Wextra -std=c99 "答案_练习2_calloc计数.c" -o a9_p2.exe * 运行：.\a9_p2.exe * * 要点： * ① calloc 一步完成「申请 + 全 0」，计数数组必须先清零 * ② malloc + memset 的等价写法：第三个参数是「总字节数」 * ③ memset(p, 0, sizeof p) 是经典陷阱：sizeof 指针只有 8 字节 * ④ 陷阱演示前先把每个元素初始化，避免读未初始化值（那是未定义行为） */ #include <stdio.h> #include <stdlib.h> #include <string.h> #define DIGITS 10 static void print_counts(const int *cnt); int main(void) { int data[] = {3, 1, 4, 1, 5, 9, 2, 6, 5, 3, 5, 8, 9, 7, 9, 3}; const int data_n = (int)(sizeof data / sizeof data[0]); int i; printf("数据序列："); for (i = 0; i < data_n; i++) { printf("%d ", data[i]); } printf("\n\n"); /* ── 1. calloc：申请 10 个 int，并保证全 0 ─────── */ printf("── calloc 统计 ──\n"); int *cnt_calloc = calloc(DIGITS, sizeof(int)); if (cnt_calloc == NULL) { printf("calloc 失败：内存不足\n"); return 1; } for (i = 0; i < data_n; i++) { cnt_calloc[data[i]]++; /* 下标就是数字本身 */ } print_counts(cnt_calloc); printf("\n"); /* ── 2. malloc + memset：等价写法 ─────────────── */ printf("── malloc + memset 统计 ──\n"); int *cnt_memset = malloc((size_t)DIGITS * sizeof(int)); if (cnt_memset == NULL) { printf("malloc 失败：内存不足\n"); free(cnt_calloc); return 1; } memset(cnt_memset, 0, (size_t)DIGITS * sizeof(int)); /* 清 DIGITS 个 int */ for (i = 0; i < data_n; i++) { cnt_memset[data[i]]++; } print_counts(cnt_memset); int same = 1; for (i = 0; i < DIGITS; i++) { if (cnt_calloc[i] != cnt_memset[i]) { same = 0; } } printf("两份统计一致：%s\n\n", same ? "OK" : "FAIL"); /* ── 3. memset 的 sizeof 陷阱（安全演示）───────── */ printf("── memset 的 sizeof 陷阱 ──\n"); int *p = malloc(DIGITS * sizeof(int)); if (p == NULL) { printf("malloc 失败\n"); free(cnt_calloc); free(cnt_memset); return 1; } for (i = 0; i < DIGITS; i++) { p[i] = 7; /* 先全部写一遍，读的时候才有定义 */ } size_t wrong_size = sizeof p; /* 陷阱就在这：这是指针的大小 8 */ memset(p, 0, wrong_size); /* 只清了前 8 字节 = 2 个 int */ printf("sizeof p = %zu 字节，10 个 int 实际需要 %zu 字节\n", sizeof p, (size_t)DIGITS * sizeof(int)); printf("错误 memset 后：p[0]=%d p[1]=%d p[2]=%d（只清了前 2 个 int）\n", p[0], p[1], p[2]); printf("正确写法：memset(p, 0, 10 * sizeof(int));\n"); /* ── 4. 收尾 ─────────────────────────────────── */ free(p); p = NULL; free(cnt_calloc); cnt_calloc = NULL; free(cnt_memset); cnt_memset = NULL; printf("三块内存已全部 free，指针已置 NULL。\n"); return 0; } static void print_counts(const int *cnt) { int d; printf("数字:次数\n"); for (d = 0; d < DIGITS; d++) { printf(" %d:%d\n", d, cnt[d]); } } /* 思考回答：malloc 拿到的是「上次别人用过的内存」，内容不确定；不清零就直接 * cnt[data[i]]++，等于在垃圾值上加一，统计结果全错。calloc 保证按位全 0， * 对 int 来说就是数值 0，所以计数数组首选 calloc。 */ /* 预期输出（本机实测）： * 数据序列：3 1 4 1 5 9 2 6 5 3 5 8 9 7 9 3 * * ── calloc 统计 ── * 数字:次数 * 0:0 * 1:2 * 2:1 * 3:3 * 4:1 * 5:3 * 6:1 * 7:1 * 8:1 * 9:3 * * ── malloc + memset 统计 ── * 数字:次数 * 0:0 * 1:2 * 2:1 * 3:3 * 4:1 * 5:3 * 6:1 * 7:1 * 8:1 * 9:3 * 两份统计一致：OK * * ── memset 的 sizeof 陷阱 ── * sizeof p = 8 字节，10 个 int 实际需要 40 字节 * 错误 memset 后：p[0]=0 p[1]=0 p[2]=7（只清了前 2 个 int） * 正确写法：memset(p, 0, 10 * sizeof(int)); * 三块内存已全部 free，指针已置 NULL。 */
CODE答案_练习4_realloc扩容.c
/* 答案_练习4_realloc扩容.c —— 阶段九练习4参考代码 * 编译（PowerShell）： * & "C:\Program Files (x86)\mingw64\bin\gcc.exe" -Wall -Wextra -std=c99 "答案_练习4_realloc扩容.c" -o a9_p4.exe * 运行：.\a9_p4.exe * * 要点： * ① capacity（租了几个格子）与 length（用了几个）分离 * ② 满了才 realloc，容量翻倍；realloc 失败时原数组必须保住 * ③ realloc 成功后旧地址可能作废：先打印旧地址，再打印新地址，别回头解引用 * ④ 空栈的 pop/top 用返回值报错，不返回垃圾值 * * 思考回答：翻倍是均摊 O(1)，从 2 涨到 1024 只需约 9 次 realloc； * 每次 +1 要 1022 次，每次都可能搬整个数组，总代价 O(n^2)。 */ #include <stdio.h> #include <stdlib.h> struct Stack { int *data; /* 堆上的格子 */ int length; /* 用了几个 */ int capacity; /* 一共租了几个 */ }; int stack_init(struct Stack *s); int stack_push(struct Stack *s, int value); int stack_pop(struct Stack *s, int *out); int stack_top(const struct Stack *s, int *out); void stack_destroy(struct Stack *s); int main(void) { struct Stack s; int i; int value; if (!stack_init(&s)) { printf("初始化失败：内存不足\n"); return 1; } printf("初始化：length = %d，capacity = %d，data = %p\n\n", s.length, s.capacity, (void *)s.data); /* ── push 1~10：观察三次扩容 ─────────────────── */ for (i = 1; i <= 10; i++) { if (!stack_push(&s, i)) { printf("push %d 失败，清理后退出\n", i); stack_destroy(&s); return 1; } printf("push %2d 后：length = %d，capacity = %d\n", i, s.length, s.capacity); } printf("\n"); /* ── pop 三次 + top 一次 ─────────────────────── */ for (i = 0; i < 3; i++) { if (stack_pop(&s, &value)) { printf("pop -> %d（剩余 length = %d）\n", value, s.length); } else { printf("pop 失败：栈空\n"); } } if (stack_top(&s, &value)) { printf("top -> %d（只看不弹，length = %d）\n", value, s.length); } else { printf("top 失败：栈空\n"); } /* ── 一直 pop 到空 ───────────────────────────── */ printf("\n清空栈：\n"); while (stack_pop(&s, &value)) { printf("pop -> %d ", value); } printf("\n已弹空：length = %d，capacity = %d\n", s.length, s.capacity); if (!stack_pop(&s, &value)) { printf("空栈 pop 返回 0：OK\n"); } if (!stack_top(&s, &value)) { printf("空栈 top 返回 0：OK\n"); } stack_destroy(&s); printf("destroy 后：data = %p，length = %d，capacity = %d\n", (void *)s.data, s.length, s.capacity); return 0; } int stack_init(struct Stack *s) { s->capacity = 2; s->length = 0; s->data = malloc((size_t)s->capacity * sizeof(int)); if (s->data == NULL) { s->capacity = 0; return 0; } return 1; } int stack_push(struct Stack *s, int value) { if (s->length == s->capacity) { /* 满了：翻倍扩容 */ int new_capacity = s->capacity * 2; printf(" [扩容] 容量 %d -> %d，旧 data = %p\n", s->capacity, new_capacity, (void *)s->data); int *tmp = realloc(s->data, (size_t)new_capacity * sizeof(int)); if (tmp == NULL) { /* 失败：原数组仍然有效 */ printf(" [扩容] realloc 失败，保留原数组\n"); return 0; } s->data = tmp; s->capacity = new_capacity; printf(" [扩容] 新 data = %p\n", (void *)s->data); } s->data[s->length] = value; s->length++; return 1; } int stack_pop(struct Stack *s, int *out) { if (s->length == 0) { return 0; /* 空栈：不动 *out */ } s->length--; if (out != NULL) { *out = s->data[s->length]; } return 1; } int stack_top(const struct Stack *s, int *out) { if (s->length == 0) { return 0; } if (out != NULL) { *out = s->data[s->length - 1]; } return 1; } void stack_destroy(struct Stack *s) { free(s->data); s->data = NULL; s->length = 0; s->capacity = 0; } /* 预期输出（本机实测；地址每次运行不同，以 0x... 示意。 * 本机第 1、3 次扩容原地，第 2 次搬家；其它机器可能不同）： * 初始化：length = 0，capacity = 2，data = 0x... * * push 1 后：length = 1，capacity = 2 * push 2 后：length = 2，capacity = 2 * [扩容] 容量 2 -> 4，旧 data = 0x... * [扩容] 新 data = 0x... * push 3 后：length = 3，capacity = 4 * push 4 后：length = 4，capacity = 4 * [扩容] 容量 4 -> 8，旧 data = 0x... * [扩容] 新 data = 0x... * push 5 后：length = 5，capacity = 8 * push 6 后：length = 6，capacity = 8 * push 7 后：length = 7，capacity = 8 * push 8 后：length = 8，capacity = 8 * [扩容] 容量 8 -> 16，旧 data = 0x... * [扩容] 新 data = 0x... * push 9 后：length = 9，capacity = 16 * push 10 后：length = 10，capacity = 16 * * pop -> 10（剩余 length = 9） * pop -> 9（剩余 length = 8） * pop -> 8（剩余 length = 7） * top -> 7（只看不弹，length = 7） * * 清空栈： * pop -> 7 pop -> 6 pop -> 5 pop -> 4 pop -> 3 pop -> 2 pop -> 1 * 已弹空：length = 0，capacity = 16 * 空栈 pop 返回 0：OK * 空栈 top 返回 0：OK * destroy 后：data = 0x0，length = 0，capacity = 0 */

/* 答案_练习5_文件统计.c —— 阶段九练习5参考代码（wc 简化版） * 编译（PowerShell）： * & "C:\Program Files (x86)\mingw64\bin\gcc.exe" -Wall -Wextra -std=c99 "答案_练习5_文件统计.c" -o a9_p5.exe * 运行：.\a9_p5.exe * * 要点： * ① 程序自己造测试文件：fputs 写完必须 fclose，否则数据可能还在缓冲区 * ② 读的时候必须 fgets 逐行，不能假设一行不超过某个长度 * ③ 单词计数用 in_word 状态机：空白收尾，非空白开头才计数 * ④ 字符数包含行尾换行符和空格，和 wc -c 的口径接近（不含 Windows 的 \r） * ⑤ 跑完 remove 测试文件，不在交付目录留 .txt */ #include <stdio.h> #include <stdlib.h> #include <string.h> #include <ctype.h> #define FILE_NAME "stage9_wc_demo.txt" int main(void) { const char *lines[5] = { "UPRE robot lab\n", "C is simple but deep\n", "\n", /* 空行：行数要算，单词数不增加 */ "dynamic memory and file IO\n", /* 连续三个空格：只能算一个分隔 */ "stage nine\n" }; int i; /* ── 1. 生成测试文件 ─────────────────────────── */ FILE *fp = fopen(FILE_NAME, "w"); if (fp == NULL) { printf("无法创建 %s\n", FILE_NAME); return 1; } for (i = 0; i < 5; i++) { if (fputs(lines[i], fp) == EOF) { printf("第 %d 行写入失败\n", i + 1); fclose(fp); remove(FILE_NAME); return 1; } } if (fclose(fp) != 0) { printf("fclose 失败，测试文件可能不完整\n"); remove(FILE_NAME); return 1; } printf("已生成测试文件 %s（5 行，含 1 个空行和连续空格）\n\n", FILE_NAME); /* ── 2. 逐行读并统计 ─────────────────────────── */ fp = fopen(FILE_NAME, "r"); if (fp == NULL) { printf("打开 %s 失败\n", FILE_NAME); return 1; } char buf[256]; int line_count = 0; int word_count = 0; long char_count = 0; while (fgets(buf, sizeof buf, fp) != NULL) { size_t len = strlen(buf); size_t j; int in_word = 0; line_count++; char_count += (long)len; for (j = 0; j < len; j++) { unsigned char ch = (unsigned char)buf[j]; if (isspace(ch)) { in_word = 0; /* 空白：当前单词结束 */ } else if (!in_word) { word_count++; /* 从空白进入非空白：新单词开始 */ in_word = 1; } } printf("第 %d 行（%zu 字符）：%s", line_count, len, buf); } fclose(fp); printf("\n统计：%d 行，%d 个单词，%ld 个字符（含行尾换行）\n", line_count, word_count, char_count); /* ── 3. 清理 ─────────────────────────────────── */ if (remove(FILE_NAME) == 0) { printf("测试文件 %s 已删除\n", FILE_NAME); } else { printf("测试文件删除失败，请手动清理 %s\n", FILE_NAME); } return 0; } /* 预期输出（本机实测）： * 已生成测试文件 stage9_wc_demo.txt（5 行，含 1 个空行和连续空格） * * 第 1 行（15 字符）：UPRE robot lab * 第 2 行（21 字符）：C is simple but deep * 第 3 行（1 字符）： * 第 4 行（29 字符）：dynamic memory and file IO * 第 5 行（11 字符）：stage nine * * 统计：5 行，15 个单词，77 个字符（含行尾换行） * 测试文件 stage9_wc_demo.txt 已删除 */

/* 答案_练习6_记录持久化.c —— 阶段九练习6参考代码 * 编译（PowerShell）： * & "C:\Program Files (x86)\mingw64\bin\gcc.exe" -Wall -Wextra -std=c99 "答案_练习6_记录持久化.c" -o a9_p6.exe * 运行：.\a9_p6.exe * * 要点： * ① CSV：fprintf 一行一条，fgets + sscanf 按逗号拆字段 * ② 读回后逐字段比对，全对才打印 ROUND TRIP OK（自检思维） * ③ 平均分用「读回来的数据」算，证明落盘再读没有丢精度 * ④ 临时文件跑完 remove，不留 .csv * * 思考回答：姓名里带逗号（如 "Smith, John"）会让 sscanf 的 %[^,] 提前截断， * 后面的分数会解析失败。正式做法是给字段加英文双引号并处理转义， * 或者干脆换用制表符/二进制格式；本题数据刻意不含逗号。 */ #include <stdio.h> #include <stdlib.h> #include <string.h> #define FILE_NAME "stage9_ex6_scores.csv" #define MAX_REC 16 struct Student { int id; char name[24]; double score; }; static int same_record(const struct Student *a, const struct Student *b); int main(void) { const struct Student origin[5] = { {7001, "Alice", 88.50}, {7002, "Bob", 76.25}, {7003, "Carol", 92.75}, {7004, "David", 59.50}, {7005, "Eve", 81.00} }; const int n = 5; int i; /* ── 1. 写 CSV ───────────────────────────────── */ printf("── 写入 CSV ──\n"); FILE *fp = fopen(FILE_NAME, "w"); if (fp == NULL) { printf("无法创建 %s\n", FILE_NAME); return 1; } for (i = 0; i < n; i++) { if (fprintf(fp, "%d,%s,%.2f\n", origin[i].id, origin[i].name, origin[i].score) < 0) { printf("第 %d 条写入失败\n", i + 1); fclose(fp); remove(FILE_NAME); return 1; } } if (fclose(fp) != 0) { printf("fclose 失败\n"); remove(FILE_NAME); return 1; } printf("已写入 %d 条到 %s\n\n", n, FILE_NAME); /* ── 2. 读回 CSV ─────────────────────────────── */ printf("── 读回 CSV ──\n"); fp = fopen(FILE_NAME, "r"); if (fp == NULL) { printf("打开 %s 失败\n", FILE_NAME); return 1; } struct Student back[MAX_REC]; int count = 0; char line[128]; while (count < MAX_REC && fgets(line, sizeof line, fp) != NULL) { struct Student tmp; int got = sscanf(line, "%d,%23[^,],%lf", &tmp.id, tmp.name, &tmp.score); if (got != 3) { printf("第 %d 行格式错误，跳过：%s", count + 1, line); continue; } back[count] = tmp; count++; } fclose(fp); printf("读回 %d 条：\n", count); for (i = 0; i < count; i++) { printf(" %d %-8s %6.2f\n", back[i].id, back[i].name, back[i].score); } /* ── 3. 往返自检 ─────────────────────────────── */ int ok = (count == n); for (i = 0; i < n && i < count; i++) { if (!same_record(&origin[i], &back[i])) { printf("第 %d 条不一致：原 %d/%s/%.2f，读回 %d/%s/%.2f\n", i + 1, origin[i].id, origin[i].name, origin[i].score, back[i].id, back[i].name, back[i].score); ok = 0; } } printf("往返自检：%s\n", ok ? "ROUND TRIP OK" : "FAIL"); /* ── 4. 用读回的数据算平均分 ─────────────────── */ double sum = 0.0; for (i = 0; i < count; i++) { sum += back[i].score; } if (count > 0) { printf("平均分 = %.2f（用读回的数据算）\n", sum / count); } /* ── 5. 清理 ─────────────────────────────────── */ if (remove(FILE_NAME) == 0) { printf("临时文件 %s 已删除\n", FILE_NAME); } else { printf("临时文件删除失败，请手动清理 %s\n", FILE_NAME); } return 0; } static int same_record(const struct Student *a, const struct Student *b) { double diff = a->score - b->score; return a->id == b->id && strcmp(a->name, b->name) == 0 && diff < 1e-9 && diff > -1e-9; } /* 预期输出（本机实测）： * ── 写入 CSV ── * 已写入 5 条到 stage9_ex6_scores.csv * * ── 读回 CSV ── * 读回 5 条： * 7001 Alice 88.50 * 7002 Bob 76.25 * 7003 Carol 92.75 * 7004 David 59.50 * 7005 Eve 81.00 * 往返自检：ROUND TRIP OK * 平均分 = 79.60（用读回的数据算） * 临时文件 stage9_ex6_scores.csv 已删除 */

/* 答案_练习7_迷你通讯录.c —— 阶段九练习7参考代码（动态数组 + 文件 + 菜单） * 编译（PowerShell）： * & "C:\Program Files (x86)\mingw64\bin\gcc.exe" -Wall -Wextra -std=c99 "答案_练习7_迷你通讯录.c" -o a9_p7.exe * 运行（交互）：.\a9_p7.exe * struct Contact * ④ fread 用「返回 1 才计数」的循环，防止文件被截断时读进垃圾 * ⑤ 数据文件 stage9_contacts.dat 是持久数据，测试完再手动删；程序不自动删 * * 思考回答：结构体加 int age 后，sizeof(struct Contact) 变了，旧文件里每条 * 记录的长度对不上，fread 会整体错位（还可能只读回半条）。所以 * *count, int *capacity) { FILE *fp = fopen(FILE_NAME, "rb"); if (fp == NULL) { printf("未找到数据文件 %s，从空通讯录开始\n", FILE_NAME); return 1; } int n = 0; if (fread(&n, sizeof n, 1, fp) != 1) { printf("文件头读取失败，按空通讯录处理\n"); fclose(fp); return 0; } if (n < 0 || n > 100000) { /* 防御：坏文件头不要拿来 malloc */ printf("记录条数异常（%d），拒绝加载\n", n); fclose(fp); return 0; } if (n == 0) { printf("数据文件里没有记录，从空通讯录开始\n"); fclose(fp); return 1; } struct Contact *tmp = malloc((size_t)n * sizeof *tmp); if (tmp == NULL) { printf("加载失败：内存不足\n"); fclose(fp); return 0; } size_t got = 0; while (got < (size_t)n && fread(&tmp[got], sizeof tmp[0], 1, fp) == 1) { got++; /* 一次读一条，返回 1 才计数 */ } fclose(fp); if (got != (size_t)n) { printf("数据文件不完整：期望 %d 条，实际读到 %zu 条\n", n, got); free(tmp); return 0; } *list = tmp; *count = n; *capacity = n; printf("已从 %s 加载 %d 条记录\n", FILE_NAME, n); return 1; } static int save_contacts(const struct Contact *list, int count) { FILE *fp = fopen(FILE_NAME, "wb"); if (fp == NULL) { printf("保存失败：无法创建 %s\n", FILE_NAME); return 0; } if (fwrite(&count, sizeof count, 1, fp) != 1) { printf("保存失败：写记录条数出错\n"); fclose(fp); remove(FILE_NAME); return 0; } if (count > 0) { size_t written = fwrite(list, sizeof *list, (size_t)count, fp); if (written != (size_t)count) { printf("保存失败：只写了 %zu / %d 条\n", written, count); fclose(fp); remove(FILE_NAME); return 0; } } if (fclose(fp) != 0) { printf("保存失败：fclose 出错\n"); return 0; } printf("已保存 %d 条到 %s\n", count, FILE_NAME); return 1; } static int ensure_capacity(struct Contact **list, int *capacity, int need) { if (need <= *capacity) { return 1; } int new_capacity = (*capacity == 0) ? 4 : *capacity; while (new_capacity < need) { new_capacity *= 2; } struct Contact *tmp = realloc(*list, (size_t)new_capacity * sizeof **list); if (tmp == NULL) { return 0; /* 原数组仍在，调用者可以继续用 */ } printf(" [扩容] 容量 %d -> %d\n", *capacity, new_capacity); *list = tmp; *capacity = new_capacity; return 1; } static void add_contact(struct Contact **list, int *count, int *capacity) { struct Contact c; printf("姓名（不含空格）：\n"); if (scanf("%31s", c.name) != 1) { printf("读取姓名失败\n"); return; } printf("电话：\n"); if (scanf("%19s", c.phone) != 1) { printf("读取电话失败\n"); return; } if (!ensure_capacity(list, capacity, *count + 1)) { printf("添加失败：内存不足\n"); return; } (*list)[*count] = c; /* 结构体整体赋值，数组字段也一起复制 */ (*count)++; printf("已添加：%s / %s（当前 %d 条）\n", c.name, c.phone, *count); } static void delete_contact(struct Contact *list, int *count) { char name[NAME_LEN]; int i; int found = -1; printf("要删除的姓名：\n"); if (scanf("%31s", name) != 1) { printf("读取姓名失败\n"); return; } for (i = 0; i < *count; i++) { if (strcmp(list[i].name, name) == 0) { found = i; break; } } if (found < 0) { printf("没找到 %s\n", name); return; } for (i = found; i < *count - 1; i++) { /* 后面的记录整体前移 */ list[i] = list[i + 1]; } (*count)--; printf("已删除 %s（剩余 %d 条）\n", name, *count); } static void search_contact(const struct Contact *list, int count) { char name[NAME_LEN]; int i; printf("要查找的姓名：\n"); if (scanf("%31s", name) != 1) { printf("读取姓名失败\n"); return; } for (i = 0; i < count; i++) { if (strcmp(list[i].name, name) == 0) { printf("找到：%s / %s\n", list[i].name, list[i].phone); return; } } printf("没找到 %s\n", name); } static void list_contacts(const struct Contact *list, int count) { int i; if (count == 0) { printf("通讯录为空\n"); return; } printf("通讯录（共 %d 条）：\n", count); for (i = 0; i < count; i++) { printf(" [%d] %-15s %s\n", i + 1, list[i].name, list[i].phone); } } static void print_menu(void) { printf("\n=== 迷你通讯录 ===\n"); printf("1. 添加联系人\n"); printf("2. 删除联系人（按姓名）\n"); printf("3. 查找联系人（按姓名）\n"); printf("4. 列出全部\n"); printf("5. 保存到文件\n"); printf("6. 退出\n"); } /*
