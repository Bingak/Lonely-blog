CODE答案_练习2_三变量轮转.c
/* 答案_练习2_三变量轮转.c —— 阶段五练习2参考代码 * 编译：gcc -Wall -Wextra 答案_练习2_三变量轮转.c -o a5_p2.exe * 要点：swap 收门牌号；rotate_right 只用两次 swap 完成右轮转。 * (a,b,c)=(1,2,3) → swap(a,b) → (2,1,3) → swap(b,c) → (2,3,1) ✗ * 正确次序：swap(b,c) → (1,3,2) → swap(a,b) → (3,1,2) ✔ * 思考回答：轮转 3 次回到原样，说明轮转是可逆的（逆操作就是向左轮转， * 等价于向右轮转两次）。每次轮转是全体状态的一个「循环置换」。 */ #include <stdio.h> void swap(int *a, int *b); void rotate_right(int *a, int *b, int *c); int main(void) { int x = 1, y = 2, z = 3; int i; for (i = 1; i <= 3; i++) { rotate_right(&x, &y, &z); printf("第 %d 次轮转：(%d, %d, %d)\n", i, x, y, z); } printf("3 次后回到 (%d, %d, %d) ← 与原样一致（自检通过）\n", x, y, z); return 0; } void swap(int *a, int *b) { int t = *a; *a = *b; *b = t; } void rotate_right(int *a, int *b, int *c) { /* 目标：(a,b,c) → (c,a,b)。先让 c 的值挪到 b，再让 b 位置的值挪到 a */ swap(b, c); /* (a,b,c) = (1,2,3) → (1,3,2) */ swap(a, b); /* (1,3,2) → (3,1,2) 目标达成 */ } /* 预期输出： * 第 1 次轮转：(3, 1, 2) * 第 2 次轮转：(2, 3, 1) * 第 3 次轮转：(1, 2, 3) * 3 次后回到 (1, 2, 3) ← 与原样一致（自检通过） */

/* 答案_练习4_指针遍历求和.c —— 阶段五练习4参考代码 * 编译：gcc -Wall -Wextra 答案_练习4_指针遍历求和.c -o a5_p4.exe * 要点：函数体内零下标；哨兵 p+n 只比较不解引用； * max_ptr 返回「最大元素的地址」，地址减首地址 = 第几个元素。 */ #include <stdio.h> int sum_ptr(const int *p, int n); const int *max_ptr(const int *p, int n); int main(void) { int a[7] = {3, 9, 1, 7, 5, 8, 2}; int n = (int)(sizeof(a) / sizeof(a[0])); printf("总和 = %d\n", sum_ptr(a, n)); const int *mp = max_ptr(a, n); /* mp 装着最大元素的门牌号 */ printf("最大值 = %d, 地址 = %p\n", *mp, (void *)mp); printf("它是第 %d 个元素（mp - a = %d）\n", (int)(mp - a) + 1, (int)(mp - a)); /* 下标从 0，第几个要 +1 */ return 0; } int sum_ptr(const int *p, int n) { int sum = 0; const int *q; for (q = p; q < p + n; q++) { /* p+n 是哨兵：可比较，不可解引用 */ sum += *q; } return sum; } const int *max_ptr(const int *p, int n) { const int *champion = p; /* 擂台法：先让第 0 格当擂主 */ const int *q; for (q = p + 1; q < p + n; q++) { if (*q > *champion) { /* 挑战者赢了就换擂主 */ champion = q; /* 记住的是地址，不是下标 */ } } return champion; } /* 预期输出（地址每次运行不同，以 0x... 示意）： * 总和 = 35 * 最大值 = 9, 地址 = 0x...14 * 它是第 2 个元素（mp - a = 1） */
CODE答案_练习5_divmod传出参数.c
/* 答案_练习5_divmod传出参数.c —— 阶段五练习5参考代码 * 编译：gcc -Wall -Wextra 答案_练习5_divmod传出参数.c -o a5_p5.exe * 要点：双通道分工——return 报成败（1 成功 / 0 失败），指针只负责传数据； * b == 0 时绝不碰 *q、*r（调用者的变量保持原样）。 * 思考回答：商本身可以是任何整数（包括 -1、0），用特殊返回值表示错误会 * 和正常结果撞车；双通道让「成败」与「数据」各走各的路，互不污染。 */ #include <stdio.h> int divmod(int a, int b, int *q, int *r); int main(void) { int shang, yu; if (divmod(17, 5, &shang, &yu)) { printf("17 / 5 = %d ... %d\n", shang, yu); } else { printf("17 / 5 计算失败\n"); } if (divmod(100, 7, &shang, &yu)) { printf("100 / 7 = %d ... %d\n", shang, yu); } else { printf("100 / 7 计算失败\n"); } shang = -999; yu = -999; /* 故意放哨兵值，验证失败时没被改 */ if (divmod(9, 0, &shang, &yu)) { printf("9 / 0 = %d ... %d\n", shang, yu); } else { printf("9 / 0 计算失败（除数为零），shang=%d yu=%d 保持原样\n", shang, yu); } return 0; } int divmod(int a, int b, int *q, int *r) { if (b == 0) { return 0; /* 失败：不动 *q、*r，只报告 */ } *q = a / b; /* 按 q 的门牌号写回商 */ *r = a % b; /* 按 r 的门牌号写回余数 */ return 1; /* 成功 */ } /* 预期输出： * 17 / 5 = 3 ... 2 * 100 / 7 = 14 ... 2 * 9 / 0 计算失败（除数为零），shang=-999 yu=-999 保持原样 */
CODE答案_练习6_指针版字符串库.c
/* 答案_练习6_指针版字符串库.c —— 阶段五练习6参考代码 * 编译：gcc -Wall -Wextra 答案_练习6_指针版字符串库.c -o a5_p6.exe * 要点：函数体内零下标；my_strcpy 结束循环后必须手动补 '\0'； * src 挂 const——复制前后源串不变有编译器背书。 * 思考回答：buf[10] 装 16 字符的串会越界写爆——my_strcpy 不知道 dest 有多大， * 责任在调用者：必须保证 dest 容量 ≥ strlen(src) + 1。 * （这就是真实库里 strcpy 被称为「危险函数」的原因。） */ #include <stdio.h> int my_strlen_v2(const char *s); void my_strcpy(char *dest, const char *src); int main(void) { char buf[20]; my_strcpy(buf, "Robot"); printf("复制结果 buf = \"%s\"\n", buf); printf("my_strlen_v2(\"Robot\") = %d\n", my_strlen_v2("Robot")); printf("my_strlen_v2(\"\") = %d\n", my_strlen_v2("")); printf("my_strlen_v2(buf) = %d\n", my_strlen_v2(buf)); /* const 保护验证：src 是字符串字面量，全程只读，复制后内容不变 */ const char *src = "UPRE"; char buf2[10]; my_strcpy(buf2, src); printf("复制后 src 仍是 \"%s\"，buf2 = \"%s\"\n", src, buf2); return 0; } int my_strlen_v2(const char *s) { const char *p = s; while (*p != '\0') { p++; } return (int)(p - s); /* 终点 - 起点 = 长度 */ } void my_strcpy(char *dest, const char *src) { while (*src != '\0') { /* 一路走到 src 的封底 */ *dest = *src; /* 逐字符搬运 */ dest++; src++; } *dest = '\0'; /* 关键：封底也要搬过去！ */ } /* 预期输出： * 复制结果 buf = "Robot" * my_strlen_v2("Robot") = 5 * my_strlen_v2("") = 0 * my_strlen_v2(buf) = 5 * 复制后 src 仍是 "UPRE"，buf2 = "UPRE" */
CODE答案_练习7_双指针逆置字符串.c
/* 答案_练习7_双指针逆置字符串.c —— 阶段五练习7（挑战）参考代码 * 编译：gcc -Wall -Wextra 答案_练习7_双指针逆置字符串.c -o a5_p7.exe * 要点：left 指首字符、right 指 '\0' 前一格；交换 *left 与 *right 后 * 双向向中间夹；空串用「先走 right 再判断」守住（right 从 s 出发， * 先走到封底再退一格——空串时 right 退到 s 前一格，循环条件自然不成立）。 */ #include <stdio.h> void swap_char(char *a, char *b); void reverse_string(char *s); int main(void) { char t1[] = "UPRE2026"; printf("逆置前：%s\n", t1); reverse_string(t1); printf("逆置后：%s\n\n", t1); char t2[] = ""; /* 空串：最容易翻车的边界 */ reverse_string(t2); printf("空串逆置后：\"%s\"（应为空串）\n", t2); char t3[] = "A"; /* 单字符 */ reverse_string(t3); printf("单字符逆置后：\"%s\"（应不变）\n", t3); char t4[] = "ab"; /* 偶数长度 */ reverse_string(t4); printf("偶数长度 ab → %s\n", t4); char t5[] = "abc"; /* 奇数长度：中间字符不动 */ reverse_string(t5); printf("奇数长度 abc → %s\n", t5); return 0; } void swap_char(char *a, char *b) { char t = *a; *a = *b; *b = t; } void reverse_string(char *s) { char *left = s; char *right = s; while (*right != '\0') { /* right 先走到封底 */ right++; } right--; /* 退一格：封底前一个字符 */ /* 空串时 right 退到 s 的前一格，此时 left > right，循环不进——安全 */ while (left < right) { /* 两指针相遇即停；奇数长度中间字符不动 */ swap_char(left, right); left++; right--; } } /* 预期输出： * 逆置前：UPRE2026 * 逆置后：6202ERPU * * 空串逆置后：""（应为空串） * 单字符逆置后："A"（应不变） * 偶数长度 ab → ba * 奇数长度 abc → cba */
