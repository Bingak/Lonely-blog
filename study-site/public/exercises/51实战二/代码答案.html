CODE答案_练习2_毫秒计时器.c
/* 答案_练习2_毫秒计时器.c —— 51 实战二练习 2 参考代码 * 编译：gcc -Wall -Wextra -std=c99 答案_练习2_毫秒计时器.c -o a51_2.exe * 运行：.\a51_2.exe * * 【模型】定时器 0 每 1ms 进一次中断，ISR 里只做 timer_tick()： * 若状态是 RUNNING 就让毫秒计数 +1，满了 60000 回绕到 0。 * 主循环处理按键（启动 / 停止 / 清零），并对显示格式做转换。 * 全程不用 delay、clock()、Sleep()：时间只由 tick 累加产生。 * * 【PC 模拟版】用一个 for 循环代表 1ms 节拍；按键用「第几个 tick 发生」的数组模拟。 * 真机 Keil：void t0_isr(void) interrupt 1 { TH0=0xFC; TL0=0x18; timer_tick(); } */ #include <stdio.h> #define MS_PER_SEC 1000UL #define WRAP_MS 60000UL /* 0.000 ~ 59.999，满 60 秒回 0 */ typedef enum { TIMER_STOPPED = 0, TIMER_RUNNING } timer_state_t; typedef enum { KEY_START = 0, KEY_STOP, KEY_CLEAR } key_t; typedef struct { timer_state_t state; unsigned long ms; /* 当前显示值，0 ~ 59999 */ } timer_t; typedef struct { unsigned long tick; /* 在第几个 1ms 节拍发生 */ key_t key; } event_t; void timer_init(timer_t *t); void timer_tick(timer_t *t); void timer_key(timer_t *t, key_t key, unsigned long tick); void timer_show(const timer_t *t, unsigned long tick); const char *state_name(timer_state_t s); const char *key_name(key_t k); int main(void) { static const event_t EVENTS[] = { { 0, KEY_START }, { 1234, KEY_STOP }, { 1500, KEY_START }, { 2000, KEY_CLEAR }, { 2500, KEY_STOP }, { 2600, KEY_CLEAR }, { 2700, KEY_START } }; timer_t t; unsigned long tick; unsigned long seg1 = 0, seg2 = 0, seg3; int e = 0; int n = (int)(sizeof EVENTS / sizeof EVENTS[0]); puts("=== 练习 2：毫秒计时器（启动 / 停止 / 清零） ==="); puts("节拍：定时器 0 每 1ms 一次；显示格式 秒.毫秒，满 60 秒回绕"); putchar('\n'); timer_init(&t); timer_show(&t, 0); for (tick = 0; tick <= 3000; tick++) { if (tick > 0) { timer_tick(&t); /* 先来一个 1ms 节拍 */ } while (e < n && EVENTS[e].tick == tick) { timer_key(&t, EVENTS[e].key, tick); e++; } if (tick == 1234) { seg1 = t.ms; } if (tick == 2500) { seg2 = t.ms; } } seg3 = t.ms; putchar('\n'); puts("【校验】各段显示值应该和运行过的 tick 数一致"); printf(" 第 1 段 0 → 1234：%lu ms（期望 1234）%s\n", seg1, seg1 == 1234 ? "，一致" : "，不一致"); printf(" 第 2 段 1500 → 2000 清零前跑了 500 拍，清零后 2000 → 2500 又跑 500 拍：%lu ms（期望 500）%s\n", seg2, seg2 == 500 ? "，一致" : "，不一致"); printf(" 第 3 段 2700 → 3000：%lu ms（期望 300）%s\n", seg3, seg3 == 300 ? "，一致" : "，不一致"); puts(" 停止后再启动是接着走、清零后重新走——这就是「状态 + 计数」分开存的意义。"); putchar('\n'); puts("【回绕】59.999 之后再来两个节拍"); t.ms = WRAP_MS - 1; timer_show(&t, 9999); timer_tick(&t); timer_show(&t, 10000); timer_tick(&t); timer_show(&t, 10001); putchar('\n'); puts("【问答题】"); puts(" · 清零不能只把显示变量置 0：内部毫秒计数才是真值，显示是它算出来的；"); puts(" 只清显示、不清计数，下一个 tick 又会把旧值加回来（或从错误的基准继续）；"); puts(" · 停止状态按清零：显示归 0，状态仍是停止，要按启动才开始走；"); puts(" · 停止状态按启动：从当前显示值接着走，不归零——想从头开始要先清零再启动。"); return 0; } void timer_init(timer_t *t) { t->state = TIMER_STOPPED; t->ms = 0; } /* 1ms 节拍：只有 RUNNING 才累加，满 60 秒回绕 */ void timer_tick(timer_t *t) { if (t->state != TIMER_RUNNING) { return; } t->ms++; if (t->ms >= WRAP_MS) { t->ms = 0; } } /* 按键处理：启动 / 停止 / 清零，每次都打印状态转移 */ void timer_key(timer_t *t, key_t key, unsigned long tick) { timer_state_t old = t->state; if (key == KEY_START) { t->state = TIMER_RUNNING; } else if (key == KEY_STOP) { t->state = TIMER_STOPPED; } else { t->ms = 0; } printf(">>> tick %4lu：%s → %s（按「%s」），显示 %lu.%03lu s\n", tick, state_name(old), state_name(t->state), key_name(key), t->ms / MS_PER_SEC, t->ms % MS_PER_SEC); } /* 打印 秒.毫秒，固定三位 */ void timer_show(const timer_t *t, unsigned long tick) { printf(" tick %6lu：%s 显示 %02lu.%03lu\n", tick, state_name(t->state), t->ms / MS_PER_SEC, t->ms % MS_PER_SEC); } const char *state_name(timer_state_t s) { return (s == TIMER_RUNNING) ? "运行" : "停止"; } const char *key_name(key_t k) { if (k == KEY_START) { return "启动"; } if (k == KEY_STOP) { return "停止"; } return "清零"; } /* 预期输出（本机实测）： * === 练习 2：毫秒计时器（启动 / 停止 / 清零） === * 节拍：定时器 0 每 1ms 一次；显示格式 秒.毫秒，满 60 秒回绕 * * tick 0：停止 显示 00.000 * >>> tick 0：停止 → 运行（按「启动」），显示 0.000 s * >>> tick 1234：运行 → 停止（按「停止」），显示 1.234 s * >>> tick 1500：停止 → 运行（按「启动」），显示 1.234 s * >>> tick 2000：运行 → 运行（按「清零」），显示 0.000 s * >>> tick 2500：运行 → 停止（按「停止」），显示 0.500 s * >>> tick 2600：停止 → 停止（按「清零」），显示 0.000 s * >>> tick 2700：停止 → 运行（按「启动」），显示 0.000 s * * 【校验】各段显示值应该和运行过的 tick 数一致 * 第 1 段 0 → 1234：1234 ms（期望 1234），一致 * 第 2 段 1500 → 2000 清零前跑了 500 拍，清零后 2000 → 2500 又跑 500 拍：500 ms（期望 500），一致 * 第 3 段 2700 → 3000：300 ms（期望 300），一致 * 停止后再启动是接着走、清零后重新走——这就是「状态 + 计数」分开存的意义。 * * 【回绕】59.999 之后再来两个节拍 * tick 9999：运行 显示 59.999 * tick 10000：运行 显示 00.000 * tick 10001：运行 显示 00.001 * * 【问答题】 * · 清零不能只把显示变量置 0：内部毫秒计数才是真值，显示是它算出来的； * 只清显示、不清计数，下一个 tick 又会把旧值加回来（或从错误的基准继续）； * · 停止状态按清零：显示归 0，状态仍是停止，要按启动才开始走； * · 停止状态按启动：从当前显示值接着走，不归零——想从头开始要先清零再启动。 */

/* 答案_练习4_串口命令解析.c —— 51 实战二练习 4 参考代码 * 编译：gcc -Wall -Wextra -std=c99 答案_练习4_串口命令解析.c -o a51_4.exe * 运行：.\a51_4.exe * * 【命令】led on / led off / beep；空格（可多个）、Tab、回车、换行都是分隔符。 * 【状态机】ST_CMD 等 led 或 beep；ST_LED_ARG 等 on 或 off。 * 字节可以跨块到（先 le 再 d on），因为 token 缓冲是跨字节保留的。 * 【PC 模拟版】用几个块模拟串口陆续收字节；真机把 feed_char 放进 RI 中断。 */ #include <stdio.h> #include <string.h> #define MAX_SEQ 8 #define TOK_MAX 16 typedef enum { ST_CMD, /* 等一条命令的开头 */ ST_LED_ARG /* 已经收到 led，等 on/off */ } parse_state_t; static parse_state_t g_state = ST_CMD; static char g_tok[TOK_MAX]; static int g_len; static int g_pos; static int g_cmds, g_errors; static int g_all_cmds, g_all_errors; static char g_seq[MAX_SEQ][TOK_MAX]; static int g_seq_n; void emit_cmd(const char *cmd); void feed_char(char ch); void handle_token(void); void run_case(const char *title, const char *const *chunks, int n); int main(void) { static const char *const C1[] = { "le", "d on" }; static const char *const C2[] = { "led on led off beep" }; static const char *const C3[] = { "led onx" }; static const char *const C4[] = { "on" }; static const char *const C5[] = { "led" }; static const char *const C6[] = { "beep led on" }; puts("=== 练习 4：串口命令解析（led on / led off / beep） ==="); puts("命令之间用空格 / Tab / 回车 / 换行分隔；led 和 on/off 之间也要空格"); putchar('\n'); run_case("跨字节：「le」+「d on」", C1, 2); run_case("一行三条命令，空格数量随意", C2, 1); run_case("led onx：on 后面粘了 x", C3, 1); run_case("只有 on：缺少 led 前缀", C4, 1); run_case("只发 led 就结束：命令不完整", C5, 1); run_case("顺序无关：beep 在前、led on 在后", C6, 1); puts("【总结】"); printf(" 6 条输入流：命令 %d 条，错误 %d 条\n", g_all_cmds, g_all_errors); puts(" 为什么不能每收一个字节就 strcmp：命令可能只到一半，必须等分隔符再判；"); puts(" 缓冲开 16 字节够放最长的「led off」和多余字符，超长要单独报错。"); return 0; } void emit_cmd(const char *cmd) { const char *desc = (strcmp(cmd, "beep") == 0) ?
